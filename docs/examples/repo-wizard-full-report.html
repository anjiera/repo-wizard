A dedicated utility (llm-guardrail.js) can be added to the scripts directory to screen prompts and generated outputs for suspicious injection patterns and system instruction disclosure attempts.

/**
 * scripts/llm-guardrail.js
 *
 * Lightweight input/output guardrail checker for AI/ML agents.
 * Screens inputs for jailbreaks and prevents system prompt leakages.
 */

'use strict';

const JAILBREAK_PATTERNS = [
  /ignore\s+(?:all\s+)?previous\s+instructions/i,
  /system\s+override/i,
  /you\s+must\s+bypass/i,
  /forget\s+your\s+(?:rules|directions|guidelines)/i
];

const LEAKAGE_PATTERNS = [
  /system\s+prompt/i,
  /underlying\s+instructions/i,
  /you\s+are\s+a\s+senior\s+ai\s+robustness/i
];

function checkInput(text) {
  for (const pattern of JAILBREAK_PATTERNS) {
    if (pattern.test(text)) {
      return { safe: false, reason: 'Suspicious jailbreak/override pattern detected' };
    }
  }
  return { safe: true };
}

function checkOutput(text) {
  for (const pattern of LEAKAGE_PATTERNS) {
    if (pattern.test(text)) {
      return { safe: false, reason: 'Potential system prompt leakage detected' };
    }
  }
  return { safe: true };
}

module.exports = {
  checkInput,
  checkOutput
};

const crypto = require('crypto');

const ALGORITHM = 'aes-256-gcm';
const IV_LENGTH = 12;

function encrypt(text, secretKey) {
  const iv = crypto.randomBytes(IV_LENGTH);
  const cipher = crypto.createCipheriv(ALGORITHM, Buffer.from(secretKey, 'hex'), iv);
  
  let encrypted = cipher.update(text, 'utf8', 'hex');
  encrypted += cipher.final('hex');
  
  const authTag = cipher.getAuthTag().toString('hex');
  return `${iv.toString('hex')}:${authTag}:${encrypted}`;
}

function decrypt(cipherText, secretKey) {
  const [ivHex, authTagHex, encryptedHex] = cipherText.split(':');
  const iv = Buffer.from(ivHex, 'hex');
  const authTag = Buffer.from(authTagHex, 'hex');
  const decipher = crypto.createDecipheriv(ALGORITHM, Buffer.from(secretKey, 'hex'), iv);
  
  decipher.setAuthTag(authTag);
  let decrypted = decipher.update(encryptedHex, 'hex', 'utf8');

  return decrypted;
}

/**
 * Executes a fetch request with exponential backoff and randomized jitter.
 */
async function fetchWithRetry(url, options = {}, retries = 3, minTimeout = 1000, factor = 2, maxTimeout = 8000) {
  let attempt = 0;
  while (attempt <= retries) {
    try {
      const response = await fetch(url, options);
      if (response.ok) {
        return response;
      }
      
      // Do not retry on client-side errors (4xx) except 429 Rate Limits
      if (response.status >= 400 && response.status < 500 && response.status !== 429) {
        throw new Error(`Client error: ${response.statusText} (${response.status})`);
      }
      
      throw new Error(`Server returned status: ${response.status}`);
    } catch (error) {
      attempt++;
      if (attempt > retries) {
        throw error;
      }
      
      // Calculate exponential backoff with randomized jitter
      const backoff = minTimeout * Math.pow(factor, attempt - 1);
      const jitter = Math.random() * 200; // randomized delay modifier
      const delay = Math.min(backoff + jitter, maxTimeout);
      
      console.warn(`[Retry Warning] Attempt ${attempt} failed. Retrying in ${Math.round(delay)}ms... Error: ${error.message}`);
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
}

2. Automated & Manual Codebase Analysis Findings

The audit evaluated static documentation files and script utilities within the workspace:

2.1 Semantic HTML & DOM Structures
  • Heading Element Nesting inside Summary Tags: In the document conversion script md-to-html.js, heading elements (<h2>, <h3>, <h4>) are wrapped directly inside <summary> tags to structure section headers. While this is valid in HTML5, some assistive technologies and screen readers may experience issues parsing nested interactive structures or announce the headers repetitively.
    • Mitigation: Consider using standard styling classes on text or using aria attributes to define heading levels on non-interactive text wrappers inside the <summary> element, rather than nesting block-level headings.
  • Language Specifications: Documentation files like getting-started.html correctly specify the language attribute on the root HTML tag (<html lang="en">), satisfying WCAG SC 3.1.1 (Language of Page).
2.2 Visual Contrast & Preferences
  • Whitepaper Style Contrast: In the md-to-html.js configuration, the whitepaper style specifies identical color schemes for both light and dark preferences (e.g. background #ffffff, text #1f2937). Users with a system preference for dark mode will not receive a dark theme under this style.
    • Mitigation: Update the dark sub-object for the whitepaper style in the script to provide dark-theme color mappings.
  • Contrast Ratios: Contrast levels under the dark-blue theme are compliant:
    • Light mode body text (#334155 on #fafafa background) yields a 10.4:1 contrast ratio.
    • Dark mode body text (#cbd5e1 on #0f172a background) yields a 9.0:1 contrast ratio.
    • Accent colors (#3b82f6 on #0f172a) yield a 4.6:1 contrast ratio, which is above the 4.5:1 Level AA threshold.
